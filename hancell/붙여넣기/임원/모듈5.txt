Private Sub LoadTables()
    Dim r As Object, k As Long, j As Long, row As Long
    Set r = NameRange("t_top")
    For k = 1 To 4
        For j = 1 To 5
            tbHead(k, j) = ""
            tbText(k, j) = ""
        Next
        tbLabel(k) = Choose(k, "인접회사활동", "휴가", "출장/방문", "근무")
        If Not r Is Nothing Then
            row = TblRow0(k)
            tbLabel(k) = CellText(r.Offset(row, 0).Value)
            For j = 1 To TblN(k)
                If TblHasHeads(k) Then tbHead(k, j) = CellText(r.Offset(row + j - 1, 2).Value)
                tbText(k, j) = CellText(r.Offset(row + j - 1, 3).Value)
            Next
        End If
    Next
End Sub

Public Sub DoBuild()
    StepTo "settings"
    LoadSettings
    StepTo "execs"
    LoadExecs
    StepTo "sync names"
    SyncExecNames
    StepTo "tables"
    LoadTables
    StepTo "measure init"
    SetScreen True
    MsInit SH_MS
    StepTo "fit"
    Dim lo As Double, hi As Double, fs As Double, ok As Boolean, need As Double, avail As Double
    Dim req() As Double, nat() As Double, headH As Double, tabH As Double, footH As Double, h() As Double
    Dim a As Long, b As Long, md As Long
    ReDim req(1 To 3)
    ReDim nat(1 To 8)
    ReDim h(1 To 3)
    gTrace = "fit lo/hi"
    lo = MinD(gMinF, gMaxF)
    hi = MaxD(gMinF, gMaxF)
    lo = R5(lo)
    hi = R5(hi)
    gTrace = "fit call"
    FitEval hi, lo, hi, ok, need, avail, req, nat, headH, tabH, footH
    If ok Then
        fs = hi
    Else
        FitEval lo, lo, hi, ok, need, avail, req, nat, headH, tabH, footH
        fs = lo
        If ok Then
            a = CLng(lo * 2)
            b = CLng(hi * 2)
            Do While b - a > 1
                md = (a + b) \ 2
                FitEval md / 2, lo, hi, ok, need, avail, req, nat, headH, tabH, footH
                If ok Then
                    a = md
                Else
                    b = md
                End If
            Loop
            fs = a / 2
            FitEval fs, lo, hi, ok, need, avail, req, nat, headH, tabH, footH
        End If
    End If
    Dim inner As Double, extra As Double, k As Long, in2 As Double, ex2 As Double, hE As Double, hF As Double
    inner = avail - 2 * gGap
    If ok Then
        If Not gRowsSet Then
            extra = MaxD(0, inner - 1 - req(1) - req(2) - req(3))
            h(1) = req(1) + extra / 4
            h(2) = req(2) + extra / 2
            h(3) = req(3) + extra / 4
        Else
            For k = 1 To 3
                h(k) = gRows(k) * (inner - 1)
            Next
        End If
    Else
        For k = 1 To 3
            h(k) = req(k)
            If gRowsSet Then h(k) = MaxD(req(k), gRows(k) * (inner - 1))
        Next
    End If
    in2 = h(2) - gGap
    If gSplit < 0 Then
        ex2 = MaxD(0, in2 - nat(5) - nat(6))
        hE = nat(5) + ex2 / 2
        hF = nat(6) + ex2 / 2
    Else
        hE = MaxD(gSplit * in2, nat(5))
        hF = MaxD(in2 - gSplit * in2, nat(6))
    End If
    h(2) = MaxD(MaxD(h(2), hE + gGap + hF), nat(4))
    StepTo "render"
    SetScreen False
    ElReset
    Dim y As Double, x As Double, w As Double, i As Long
    y = LayHead(True)
    For i = 1 To 3
        SeatGeom i, x, w
        LayBox i, x, y, w, h(1), True
    Next
    y = y + h(1) + gGap
    SeatGeom 4, x, w
    LayBox 4, x, y, w, h(2), True
    SeatGeom 5, x, w
    LayBox 5, x, y, w, hE, True
    SeatGeom 6, x, w
    LayBox 6, x, y + hE + gGap, w, h(2) - hE - gGap, True
    y = y + h(2) + gGap
    SeatGeom 7, x, w
    LayBox 7, x, y, w, h(3), True
    SeatGeom 8, x, w
    LayBox 8, x, y, w, h(3), True
    y = y + h(3)
    If gShowTables Then
        y = y + gGap
        y = y + LayTables(y, True)
    End If
    If ok And gFoot <> "" Then y = gH - footH
    LayFoot y, True
    Dim ws As Object, lastR As Long, lastC As Long
    Set ws = HostBook().Worksheets(SH_A4)
    StepTo "realize"
    Realize ws, gFont, lastR, lastC
    StepTo "page setup"
    PageSetupA4 ws, gLandscape, gMargin, lastR, lastC
    StepTo "status"
    Dim st As String
    If ok Then
        st = "A4 한 장 자동 맞춤 · 글자 " & NumStr(fs) & "pt"
        If Not gRowsSet And gSplit < 0 And fs >= hi Then st = st & " · 최대 글자크기 · 여유 " & Format$((avail - need) / 2.834645669, "0") & "mm (칸을 늘려 채움)"
        If gRowsSet Or gSplit >= 0 Then st = st & " · 칸 높이 직접 조정"
    Else
        st = "A4 한 장을 " & Format$(-Int(-(need - avail) / 2.834645669), "0") & "mm 넘습니다 (글자 " & NumStr(fs) & "pt). "
        If gRowsSet Or gSplit >= 0 Then
            st = st & "높이를 직접 정한 칸에 내용이 넘칩니다. 서식 시트에서 [높이만 자동으로]를 누르세요."
        Else
            st = st & "내용을 줄이거나 서식 시트에서 최소 글자크기·여백을 줄여 보세요. 지금 인쇄하면 한 장에 맞게 줄여 찍힙니다."
        End If
    End If
    ShowStatus st, ok
    StepTo "sizes"
    UpdateSizes h, hE, hF
    StepTo "map"
    UpdateMap
    StepTo "a4 buttons"
    PlaceA4Buttons ws, lastC
    StepTo "done"
End Sub

Public Sub BuildA4()

End Sub

Public Sub GroupsReset()
    gN = 0
    ReDim gLabel(1 To 1)
    ReDim gTitleG(1 To 1)
    ReDim gSource(1 To 1)
    ReDim gHasExec(1 To 1)
    ReDim gHasRoles(1 To 1)
    ReDim gTarget(1 To 1)
    ReDim gRowsC(1 To 1)
    ReDim gRoles(1 To 1)
    ReDim gLabels(1 To 1)
End Sub

Private Function GroupAdd(ByVal label As String, ByVal title As String, ByVal source As String, ByVal hasExec As Boolean, ByVal hasRoles As Boolean, roles As Variant, labels As Variant) As Long
    gN = gN + 1
    ReDim Preserve gLabel(1 To gN)
    ReDim Preserve gTitleG(1 To gN)
    ReDim Preserve gSource(1 To gN)
    ReDim Preserve gHasExec(1 To gN)
    ReDim Preserve gHasRoles(1 To gN)
    ReDim Preserve gTarget(1 To gN)
    ReDim Preserve gRowsC(1 To gN)
    ReDim Preserve gRoles(1 To gN)
    ReDim Preserve gLabels(1 To gN)
    gLabel(gN) = label
    gTitleG(gN) = title
    gSource(gN) = source
    gHasExec(gN) = hasExec
    gHasRoles(gN) = hasRoles
    gTarget(gN) = -1
    gRowsC(gN) = ""
    gRoles(gN) = Join(roles, "|~|")
    gLabels(gN) = Join(labels, "|~|")
    GroupAdd = gN
End Function

Public Function SkipSheetName(ByVal nm As String) As Boolean
    Dim k As String, p As Variant, i As Long
    k = LCase$(RemoveChars(nm, ""))
    p = Array("작성방법", "작성요령", "작성예시", "안내", "도움말", "사용법", "예시", "목록", "설정", "readme", "guide", "help")
    For i = 0 To UBound(p)
        If StartsWith(k, CStr(p(i))) Then
            SkipSheetName = True
            Exit Function
        End If
    Next
End Function

Public Sub SheetGroups(ByVal sheetName As String, v As Variant, ByVal r0 As Long, ByVal source As String)
    Dim nR As Long, nC As Long, r As Long, c As Long, hr As Long, i As Long
    If Not IsArray(v) Then Exit Sub
    nR = UBound(v, 1)
    nC = UBound(v, 2)
    Dim rowS() As String
    ReDim rowS(0 To nC - 1)
    Dim roles() As String, labels() As String, hasRoles As Boolean
    ReDim roles(0 To nC - 1)
    ReDim labels(0 To nC - 1)
    hr = 0
    For r = r0 To MinD(nR, r0 + 19)
        For c = 1 To nC
            rowS(c - 1) = CellText(v(r, c))
        Next
        If IsHeaderRow(rowS) Then
            hr = r
            Exit For
        End If
    Next
    Dim ce As Long, cp As Long, cd As Long, title As String, rowTxt As String
    ce = -1
    cp = -1
    cd = -1
    If hr > 0 Then
        hasRoles = True
        For c = 1 To nC
            HeadRole CellText(v(hr, c)), roles(c - 1), labels(c - 1)
            If roles(c - 1) = "exec" And ce < 0 Then ce = c - 1
            If roles(c - 1) = "pos" And cp < 0 Then cp = c - 1
            If roles(c - 1) = "date" And cd < 0 Then cd = c - 1
        Next
        For r = r0 To hr - 1
            rowTxt = ""
            For c = 1 To nC
                If CellText(v(r, c)) <> "" Then rowTxt = TrimX(rowTxt & " " & CellText(v(r, c)))
            Next
            If rowTxt <> "" Then title = TrimX(title & " " & rowTxt)
        Next
    Else
        hr = r0 - 1
    End If
    Dim curExec As String, curPos As String, curDate As String, anyV As Boolean, key As String, g As Long, y0 As Long
    Dim keys() As String, keyG() As Long, nKeys As Long
    ReDim keys(1 To nR + 1)
    ReDim keyG(1 To nR + 1)
    y0 = CLng(Left$(gDate, 4))
    For r = hr + 1 To nR
        anyV = False
        ReDim rowS(0 To nC - 1)
        For c = 1 To nC
            rowS(c - 1) = CellText(v(r, c), hasRoles And roles(c - 1) = "time")
            If hasRoles Then
                If roles(c - 1) = "date" And rowS(c - 1) <> "" Then rowS(c - 1) = DateValueIso(v(r, c), y0)
            End If
            If rowS(c - 1) <> "" Then anyV = True
        Next
        If anyV Then
            If Not IsHeaderRow(rowS) Then
                If ce >= 0 Then
                    If rowS(ce) <> "" Then
                        curExec = rowS(ce)
                        If cp >= 0 Then curPos = rowS(cp)
                    ElseIf cp >= 0 Then
                        If rowS(cp) <> "" Then curPos = rowS(cp)
                    End If
                ElseIf cp >= 0 Then
                    If rowS(cp) <> "" Then curPos = rowS(cp)
                End If
                If cd >= 0 Then
                    If rowS(cd) <> "" Then
                        If IsIso(rowS(cd)) Then
                            curDate = rowS(cd)
                        Else
                            curDate = IIf(ParseLooseDate(rowS(cd), y0) <> "", ParseLooseDate(rowS(cd), y0), curDate)
                        End If
                    End If
                End If
                If ce >= 0 Then
                    key = "e:" & curPos & "|" & curExec
                Else
                    key = "s"
                End If
                g = FindKey(keys, nKeys, key)
                If g > 0 Then g = keyG(g)
                If g = 0 Then
                    If ce >= 0 Then
                        g = GroupAdd(TrimX(curPos & " " & curExec), title, source & IIf(TrimX(curPos & " " & curExec) <> "", " › " & TrimX(curPos & " " & curExec), ""), True, hasRoles, roles, labels)
                    Else
                        g = GroupAdd(sheetName, title, source, False, hasRoles, roles, labels)
                    End If
                    nKeys = nKeys + 1
                    keys(nKeys) = key
                    keyG(nKeys) = g
                End If
                gRowsC(g) = gRowsC(g) & SepA() & IIf(cd >= 0, curDate, "") & SepB() & Join(rowS, "|~|")
            End If
        End If
    Next
End Sub

Private Function NormKey(ByVal s As String) As String
    NormKey = LCase$(RemoveChars(RemoveParens(s), "·.,_-"))
End Function

Public Function MatchExec(ByVal label As String) As Long
    Dim L As String, i As Long, n As String, p As String, sc As Long, best As Long, bestScore As Long, tie As Boolean
    L = NormKey(label)
    best = -1
    If L = "" Then
        MatchExec = -1
        Exit Function
    End If
    For i = 1 To 8
        n = NormKey(exName(i))
        p = NormKey(exPos(i))
        sc = 0
        If n <> "" And InStr(L, n) > 0 Then
            sc = 10
            If p <> "" And InStr(L, p) > 0 Then sc = 12
        ElseIf p <> "" And L = p Then
            sc = 5
        ElseIf p <> "" And InStr(L, p) > 0 Then
            sc = 3
        End If
        If sc > bestScore Then
            bestScore = sc
            best = i
            tie = False
        ElseIf sc > 0 And sc = bestScore Then
            tie = True
        End If
    Next
    If tie Then
        MatchExec = -1
    Else
        MatchExec = best
    End If
End Function

Public Sub GroupLines(ByVal g As Long, ByVal D As String, ByRef sched As String, ByRef refc As String)
    Dim rw As Variant, cells() As String, roles() As String, labels() As String, ra As Variant, n As Long, q As Long
    roles = Split(gRoles(g), "|~|")
    labels = Split(gLabels(g), "|~|")
    n = LCount(gRowsC(g))
    If n = 0 Then Exit Sub
    ra = LItems(gRowsC(g))
    For q = 0 To n - 1
        rw = Split(CStr(ra(q)), SepB())
        If CStr(rw(0)) = "" Or D = "" Or CStr(rw(0)) = D Then
            cells = Split(CStr(rw(1)), "|~|")
            RowLines cells, roles, labels, gHasRoles(g), sched, refc
        End If
    Next
End Sub

Public Function SheetValues(ByVal ws As Object) As Variant

Public Sub ClearExecRows(ByVal ws As Object)
    Dim lastR As Long, lastC As Long
    UsedLast ws, lastR, lastC
    If lastR < 4 Then lastR = 4
    If lastC < 4 Then lastC = 4
    ws.Range(ws.Cells(4, 1), ws.Cells(lastR, lastC)).ClearContents
End Sub

Private Function AttachPlace(ByVal content As String, ByVal place As String) As String
    Dim ls As Variant, i As Long
    ls = Split(content, vbLf)
    For i = 0 To UBound(ls)
        If TrimX(CStr(ls(i))) <> "" Then
            If FirstIn(TrimX(CStr(ls(i))), "*＊※") Then
                AttachPlace = "@" & place & vbLf & content
            Else
                ls(i) = CStr(ls(i)) & " @" & place
                AttachPlace = Join(ls, vbLf)
            End If
            Exit Function
        End If
    Next
    AttachPlace = "@" & place
End Function

Private Sub WriteExecRow(ByVal ws As Object, ByRef rowN As Long, ByVal a As String, ByVal b As String, ByVal c As String, ByVal d As String)
    If a = "" And b = "" And c = "" And d = "" Then Exit Sub
    ws.Range(ws.Cells(rowN, 1), ws.Cells(rowN, 4)).NumberFormat = "@"
    ws.Cells(rowN, 1).Value = a
    ws.Cells(rowN, 2).Value = b
    ws.Cells(rowN, 3).Value = c
    ws.Cells(rowN, 4).Value = d
    rowN = rowN + 1
End Sub

Public Sub WriteGroupRows(ByVal g As Long, ByVal D As String, ByVal ws As Object, ByRef rowN As Long)
    Dim rw As Variant, cells() As String, roles() As String, labels() As String, i As Long
    Dim tm As String, content As String, place As String, stars As String, refs As String
    Dim sched As String, refc As String, ra As Variant, n As Long, q As Long, la As Variant, nl As Long, j As Long
    roles = Split(gRoles(g), "|~|")
    labels = Split(gLabels(g), "|~|")
    n = LCount(gRowsC(g))
    If n = 0 Then Exit Sub
    ra = LItems(gRowsC(g))
    For q = 0 To n - 1
        rw = Split(CStr(ra(q)), SepB())
        If CStr(rw(0)) = "" Or D = "" Or CStr(rw(0)) = D Then
            cells = Split(CStr(rw(1)), "|~|")
            If gHasRoles(g) Then
                tm = ""
                content = ""
                place = ""
                stars = ""
                refs = ""
                For i = LBound(cells) To UBound(cells)
                    If cells(i) <> "" Then
                        Select Case roles(i)
                            Case "time"
                                tm = cells(i)
                            Case "content"
                                content = content & IIf(content = "", "", vbLf) & cells(i)
                            Case "place"
                                place = cells(i)
                            Case "ref"
                                refs = refs & IIf(refs = "", "", vbLf) & cells(i)
                            Case "star"
                                If labels(i) <> "" Then
                                    stars = stars & IIf(stars = "", "", vbLf) & labels(i) & ": " & cells(i)
                                Else
                                    stars = stars & IIf(stars = "", "", vbLf) & cells(i)
                                End If
                        End Select
                    End If
                Next
                If place <> "" Then content = AttachPlace(content, place)
                WriteExecRow ws, rowN, tm, content, stars, refs
            Else
                sched = ""
                refc = ""
                RowLines cells, roles, labels, False, sched, refc
                nl = LCount(sched)
                If nl > 0 Then
                    la = LItems(sched)
                    For j = 0 To nl - 1
                        WriteExecRow ws, rowN, "", CStr(la(j)), "", ""
                    Next
                End If
                nl = LCount(refc)
                If nl > 0 Then
                    la = LItems(refc)
                    For j = 0 To nl - 1
                        WriteExecRow ws, rowN, "", "", "", CStr(la(j))
                    Next
                End If
            End If
        End If
    Next
End Sub

Public Sub Check5()
    Dim n As Long, msg As String
    msg = "5번 모듈을 읽었습니다 (전체 7개)."
    On Error Resume Next
    n = ModOk1() + ModOk2() + ModOk3() + ModOk4() + ModOk5()
    If Err.Number <> 0 Or n <> 15 Then
        msg = msg & vbLf & "그런데 앞 모듈을 부르지 못했습니다: " & Err.Description
    Else
        msg = msg & vbLf & "다음: 6번 파일을 새 모듈에 붙여넣으세요."
    End If
    On Error GoTo 0
    MsgBox msg, 0, "확인"
End Sub
