Private Sub Lbl(ByVal ws As Object, ByVal addr As String, ByVal txt As String, Optional ByVal bold As Boolean = False, Optional ByVal size As Double = 10, Optional ByVal color As Long = 0)
    With ws.Range(addr)
        If .Cells.Count > 1 Then .Merge
        .Cells(1, 1).Value = txt
        .Font.Bold = bold
        .Font.Size = size
        .Font.Color = color
        .VerticalAlignment = -4108
    End With
End Sub

Private Sub Inp(ByVal ws As Object, ByVal addr As String, ByVal v As Variant, ByVal nm As String, Optional ByVal textFmt As Boolean = True)
    With ws.Range(addr)
        If .Cells.Count > 1 Then .Merge
        If textFmt Then .NumberFormat = "@"
        .Cells(1, 1).Value = v
        .Interior.Color = 16777215
        .Borders.LineStyle = 1
        .Borders.Color = 14011590
        .VerticalAlignment = -4108
        .Font.Size = 10
    End With
    If nm <> "" Then DefName nm, ws, ws.Range(addr).Cells(1, 1).Address
End Sub

Private Sub ListValid(ByVal ws As Object, ByVal addr As String, ByVal items As String)
    On Error Resume Next
    With ws.Range(addr).Validation
        .Delete
        .Add 3, 1, 1, items
        .IgnoreBlank = True
        .InCellDropdown = True
    End With
    On Error GoTo 0
End Sub

Public Sub SetupAll()
    Dim wb As Object, ws As Object, k As Long, i As Long, nAll As Long, samp As Variant
    Set wb = HostBook()
    Application.DisplayAlerts = False
    On Error Resume Next
    wb.Styles("Normal").Font.Name = "맑은 고딕"
    wb.Styles("Normal").Font.Size = 10
    On Error GoTo 0
    StepTo "setup start"
    Do While wb.Worksheets.Count > 1
        wb.Worksheets(wb.Worksheets.Count).Delete
    Loop
    nAll = 8 + 8
    StepTo "add sheets"
    Do While wb.Worksheets.Count < nAll
        Set ws = NewSheet(wb)
    Loop
    StepTo "clear sheets"
    For i = 1 To nAll
        Set ws = wb.Worksheets(i)
        ClearSheet ws
        On Error Resume Next
        Do While ws.Shapes.Count > 0
            ws.Shapes(1).Delete
            If Err.Number <> 0 Then Exit Do
        Loop
        On Error GoTo 0
        ws.Name = "~t" & i
    Next
    StepTo "rename"
    wb.Worksheets(1).Name = SH_IN
    wb.Worksheets(2).Name = SH_A4
    For k = 1 To 8
        Set ws = wb.Worksheets(2 + k)
        StepTo "exec sheet " & k
        samp = SampleExec(k)
        SetupExecSheet ws, samp(0), samp(1), samp(2)
        ws.Name = SafeSheetName(samp(0) & " " & samp(1))
    Next
    StepTo "tables"
    Set ws = wb.Worksheets(8 + 3)
    ws.Name = SH_TBL
    SetupTables ws
    StepTo "format"
    Set ws = wb.Worksheets(8 + 4)
    ws.Name = SH_FMT
    SetupFormat ws
    StepTo "help"
    Set ws = wb.Worksheets(8 + 5)
    ws.Name = SH_HELP
    SetupHelp ws
    StepTo "paste"
    Set ws = wb.Worksheets(8 + 6)
    ws.Name = SH_PASTE
    SetupPaste ws
    StepTo "scratch"
    Set ws = wb.Worksheets(8 + 7)
    ws.Name = SH_MS
    ws.Visible = 0
    Set ws = wb.Worksheets(8 + 8)
    ws.Name = SH_BAK
    ws.Visible = 0
    StepTo "input"
    SetupInput wb.Worksheets(SH_IN)
    StepTo "a4"
    SetupA4 wb.Worksheets(SH_A4)
    Application.DisplayAlerts = True
    StepTo "build"
    DoBuild
    wb.Worksheets(SH_IN).Activate
End Sub

Public Sub Setup()
    Dim ws As Object
    On Error GoTo fail
    StepTo "setup: check"
    Set ws = SheetByName(SH_IN)
    If Not ws Is Nothing Then
        If Not Confirm("이미 만들어진 파일입니다. 처음 상태(예시 내용)로 다시 만들까요? 지금 내용은 지워집니다.") Then Exit Sub
    End If
    BeginWork
    SetupAll
    EndWork
    Info "임원 일일 예정사항 파일을 만들었습니다. [다른 이름으로 저장]에서 '매크로 사용 문서(.xlsm)'로 저장하세요."
    Exit Sub
fail:
    EndWork
    Info "파일을 만들다가 문제가 생겼습니다." & vbLf & Err.Number & " " & Err.Description & vbLf & "(단계: " & gStep & ")"
End Sub

Public Sub Check7()
    Dim n As Long, msg As String
    msg = "7번 모듈을 읽었습니다 (전체 7개)."
    On Error Resume Next
    n = ModOk1() + ModOk2() + ModOk3() + ModOk4() + ModOk5() + ModOk6() + ModOk7()
    If Err.Number <> 0 Or n <> 28 Then
        msg = msg & vbLf & "그런데 앞 모듈을 부르지 못했습니다: " & Err.Description
    Else
        msg = msg & vbLf & "모든 모듈을 넣었습니다. 이제 Setup 을 실행하세요."
    End If
    On Error GoTo 0
    MsgBox msg, 0, "확인"
End Sub
