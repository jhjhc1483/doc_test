Private Sub NormFracs(ByRef dst() As Double, ByVal n As Long, vals As Variant, defs As Variant)
    Dim i As Long, s As Double, ok As Boolean
    ok = True
    For i = 1 To n
        If vals(i - 1) <= 0 Then ok = False
    Next
    For i = 1 To n
        If ok Then
            dst(i) = vals(i - 1)
        Else
            dst(i) = defs(i - 1)
        End If
        s = s + dst(i)
    Next
    For i = 1 To n
        dst(i) = dst(i) / s
    Next
End Sub

Public Sub LoadSettings()
    gLandscape = (NStr("s_orient") = U("A4 \uAC00\uB85C"))
    gMargin = NNum("s_margin", 10, 5, 25)
    gFont = NStr("s_font")
    If gFont = "" Then gFont = U("\uB9D1\uC740 \uACE0\uB515")
    NormFracs gC1, 3, Array(NOptNum("s_c1a"), NOptNum("s_c1b"), NOptNum("s_c1c")), Array(1, 1, 1)
    NormFracs gC2, 2, Array(NOptNum("s_c2a"), NOptNum("s_c2b")), Array(2, 1)
    NormFracs gC3, 2, Array(NOptNum("s_c3a"), NOptNum("s_c3b")), Array(1, 1)
    Dim h1 As Double, h2 As Double, h3 As Double
    h1 = NOptNum("s_h1"): h2 = NOptNum("s_h2"): h3 = NOptNum("s_h3")
    gRowsSet = (h1 > 0 And h2 > 0 And h3 > 0)
    If gRowsSet Then
        gRows(1) = h1 / (h1 + h2 + h3): gRows(2) = h2 / (h1 + h2 + h3): gRows(3) = h3 / (h1 + h2 + h3)
    End If
    gSplit = NOptNum("s_split")
    If gSplit < 0.1 Or gSplit > 0.9 Then gSplit = -1
    gMaxF = NNum("s_max", 12, 6, 20)
    gMinF = NNum("s_min", 7, 5, 16)
    gShowTables = NBool("s_tables", True)
    gShowSecret = NBool("s_secret", True)
    gSecretText = NStr("s_secretText")
    If gSecretText = "" Then gSecretText = U("\uBE44\uACF5\uAC1C")
    gEmptyText = NStr("s_empty")
    gTitle = NStr("d_title")
    gDate = DateValueIso(NVal("d_date"))
    If gDate = "" Then gDate = IsoOf(Date)
    gSub = NStr("d_sub")
    gFoot = NStr("d_foot")
    If gLandscape Then
        gW = (297 - 2 * gMargin) * PT_MM
        gH = (210 - 2 * gMargin) * PT_MM
    Else
        gW = (210 - 2 * gMargin) * PT_MM
        gH = (297 - 2 * gMargin) * PT_MM
    End If
    gGap = 2.4 * PT_MM
End Sub

Private Function IsExecSheet(ByVal ws As Object) As Boolean
    On Error Resume Next
    IsExecSheet = (CStr(ws.Cells(1, 1).Value) = U("\uC9C1\uC704") And CStr(ws.Cells(1, 3).Value) = U("\uC131\uBA85"))
End Function

Private Function ExecSheetList() As Collection
    Dim c As New Collection, ws As Object
    For Each ws In HostBook().Worksheets
        If IsExecSheet(ws) Then
            If c.Count < NX Then c.Add ws
        End If
    Next
    Set ExecSheetList = c
End Function

Public Sub LoadExecs()
    Dim c As Collection, k As Long, nws As Object, after As Object
    Dim sched As Collection, refc As Collection, res As Variant, r As Variant, itc As Collection, rfc As Collection
    gStep = "execs: list"
    Set c = ExecSheetList()
    Do While c.Count < NX
        If c.Count > 0 Then
            Set after = c(c.Count)
        Else
            Set after = HostBook().Worksheets(HostBook().Worksheets.Count)
        End If
        Set nws = HostBook().Worksheets.Add(, after)
        SetupExecSheet nws, "", "", Empty
        nws.Name = UniqueSheetName(U("\uC784\uC6D0") & (c.Count + 1), nws)
        Set c = ExecSheetList()
    Loop
    For k = 1 To NX
        gStep = "execs: sheet " & k
        Set exWs(k) = c(k)
        exPos(k) = CellText(exWs(k).Cells(1, 2).Value)
        exName(k) = CellText(exWs(k).Cells(1, 4).Value)
        Set sched = New Collection
        Set refc = New Collection
        gStep = "execs: read " & k
        ReadExecRows exWs(k), sched, refc
        gStep = "execs: parse " & k
        res = ParseSched(sched)
        gStep = "execs: refs " & k
        Set itc = res(0)
        Set rfc = res(1)
        For Each r In rfc
            refc.Add r
        Next
        Set exItems(k) = itc
        Set exRefs(k) = refc
    Next
End Sub

Private Function UniqueSheetName(ByVal want As String, ByVal self As Object) As String
    Dim nm As String, n As Long, ws As Object, clash As Boolean, selfName As String
    If Not self Is Nothing Then selfName = self.Name
    want = SafeSheetName(want)
    If want = "" Then want = U("\uC784\uC6D0")
    nm = want
    n = 2
    Do
        clash = False
        For Each ws In HostBook().Worksheets
            If ws.Name <> selfName Then
                If LCase$(ws.Name) = LCase$(nm) Then clash = True
            End If
        Next
        If Not clash Then Exit Do
        nm = Left$(want, 27) & "(" & n & ")"
        n = n + 1
    Loop
    UniqueSheetName = nm
End Function

    On Error Resume Next
    For k = 1 To NX

        If want = "" Then want = U("\uC784\uC6D0") & k

Public Function UsedLast(ByVal ws As Object, ByRef lastR As Long, ByRef lastC As Long) As Boolean
    Dim ur As Object
    Set ur = ws.UsedRange
    lastR = ur.Row + ur.Rows.Count - 1
    lastC = ur.Column + ur.Columns.Count - 1
    UsedLast = True
End Function

Private Sub ReadExecRows(ByVal ws As Object, ByVal sched As Collection, ByVal refc As Collection)
    Dim lastR As Long, lastC As Long, v As Variant, r As Long, c As Long
    UsedLast ws, lastR, lastC
    If lastC < 4 Then lastC = 4
    If lastR < 4 Then Exit Sub
    v = ws.Range(ws.Cells(1, 1), ws.Cells(lastR, lastC)).Value
    Dim roles() As String, labels() As String, cells() As String, anyV As Boolean
    ReDim roles(0 To lastC - 1): ReDim labels(0 To lastC - 1): ReDim cells(0 To lastC - 1)
    For c = 1 To lastC
        HeadRole CellText(v(3, c)), roles(c - 1), labels(c - 1)
    Next
    For r = 4 To lastR
        anyV = False
        For c = 1 To lastC
            cells(c - 1) = CellText(v(r, c), roles(c - 1) = "time")
            If cells(c - 1) <> "" Then anyV = True
        Next
        If anyV Then
            If Not IsHeaderRow(cells) Then RowLines cells, roles, labels, True, sched, refc
        End If
    Next
End Sub

Private Function InList(ByVal k As String, ByVal list As String) As Boolean
    InList = (InStr("|" & list & "|", "|" & k & "|") > 0)
End Function

Public Sub HeadRole(ByVal h As String, ByRef role As String, ByRef label As String)
    Dim k As String
    label = ""
    k = LCase$(RemoveChars(RemoveParens(h), ""))
    If k = "" Then
        role = "skip"
    ElseIf InList(k, U("\uC2DC\uAC04|\uC2DC\uAC01|\uC77C\uC2DC|time")) Then
        role = "time"
    ElseIf InList(k, U("\uB0B4\uC6A9|\uC77C\uC815|\uC77C\uC815\uB0B4\uC6A9|\uC8FC\uC694\uC77C\uC815|\uC8FC\uC694\uB0B4\uC6A9|\uC608\uC815\uC0AC\uD56D|\uC8FC\uC694\uC608\uC815\uC0AC\uD56D|\uD589\uC0AC|\uD589\uC0AC\uBA85|\uC5C5\uBB34|content")) Then
        role = "content"
    ElseIf InList(k, U("\uC7A5\uC18C|\uC704\uCE58|place")) Then
        role = "place"
    ElseIf InList(k, U("\u203B|\uCC38\uACE0|\uCC38\uACE0\uC0AC\uD56D|\u203B\uCC38\uACE0|\u203B\uCC38\uACE0\uC0AC\uD56D")) Then
        role = "ref"
    ElseIf InList(k, U("\uBE44\uACE0|\uC124\uBA85|\uBD80\uAC00\uC124\uBA85|\uBA54\uBAA8|note")) Then
        role = "star"
    ElseIf InList(k, U("\uC131\uBA85|\uC774\uB984|\uC784\uC6D0|\uC784\uC6D0\uBA85|name")) Then
        role = "exec"
    ElseIf InList(k, U("\uC9C1\uC704|\uC9C1\uCC45|\uC9C1\uAE09")) Then
        role = "pos"
    ElseIf InList(k, U("\uB0A0\uC9DC|\uC77C\uC790|date")) Then
        role = "date"
    ElseIf InList(k, U("\uC21C\uBC88|\uBC88\uD638|no|no.|\uC5F0\uBC88")) Then
        role = "skip"
    Else
        role = "star"
        label = TrimX(h)
    End If
End Sub

Public Function IsHeaderRow(cells() As String) As Boolean
    Dim i As Long, r As String, lb As String, hasC As Boolean, hasT As Boolean
    For i = LBound(cells) To UBound(cells)
        HeadRole cells(i), r, lb
        If r = "content" Then hasC = True
        If r = "time" Or r = "exec" Then hasT = True
    Next
    IsHeaderRow = hasC And hasT
End Function

'
Public Sub RowLines(cells() As String, roles() As String, labels() As String, ByVal hasRoles As Boolean, _
        ByVal sched As Collection, ByVal refc As Collection)
    Dim tm As String, content As String, place As String, stars As New Collection, i As Long, c As String, l As Variant
    Dim ro As String, n As Long, st As Long
    If hasRoles Then
        For i = LBound(cells) To UBound(cells)
            c = cells(i)
            If c <> "" Then
                ro = "star"
                If i <= UBound(roles) Then ro = roles(i)
                Select Case ro
                    Case "time": tm = c
                    Case "content"
                        If content <> "" Then content = content & vbLf
                        content = content & c
                    Case "place": place = c
                    Case "ref"
                        For Each l In Split(c, vbLf)
                            If TrimX(CStr(l)) <> "" Then refc.Add StripLead(TrimX(CStr(l)), U("\u203B"))
                        Next
                    Case "star"
                        If i <= UBound(labels) And labels(i) <> "" Then
                            stars.Add labels(i) & ": " & c
                        Else
                            stars.Add c
                        End If
                End Select
            End If
        Next
    Else
        n = UBound(cells)
        Do While n >= LBound(cells)
            If cells(n) <> "" Then Exit Do
            n = n - 1
        Loop
        st = LBound(cells)
        If n - st + 1 >= 2 Then
            If cells(st) = "" Or IsTimeCell(cells(st)) Then
                tm = cells(st)
                st = st + 1
            End If
        End If
        If st <= n Then
            content = cells(st)
            st = st + 1
        End If
        For i = st To n
            If cells(i) <> "" Then stars.Add cells(i)
        Next
    End If
    Dim lines As New Collection, first As String, t As String
    For Each l In Split(content, vbLf)
        t = TrimX(CStr(l))
        If t <> "" Then lines.Add t
    Next
    If lines.Count > 0 Then
        first = lines(1)
        lines.Remove 1
        If FirstIn(first, U("*\uFF0A\u203B")) Then
            If lines.Count > 0 Then
                lines.Add first, , 1
            Else
                lines.Add first
            End If
            first = ""
        End If
    End If
    Dim tmS As String, sp As String, rest As String, ln As String
    If tm <> "" Then
        If SplitTime(tm, sp, rest) And rest = "" Then
            tmS = sp & " "
        Else
            tmS = tm & vbTab
        End If
    End If
    If first <> "" Or tmS <> "" Then
        ln = tmS & first
        If place <> "" Then ln = ln & " @" & place
        If Right$(ln, 1) = " " Then ln = Left$(ln, Len(ln) - 1)
        sched.Add ln
    ElseIf place <> "" Then
        sched.Add "@" & place
    End If
    For Each l In lines
        t = CStr(l)
        If Left$(t, 1) = U("\u203B") Then
            refc.Add StripLead(t, U("\u203B"))
        ElseIf FirstIn(t, U("*\uFF0A-\u00B7\u2022")) Then
            sched.Add t
        Else
            sched.Add "  " & t
        End If
    Next
    Dim s As Variant
    For Each s In stars
        For Each l In Split(CStr(s), vbLf)
            t = TrimX(CStr(l))
            If t <> "" Then
                If FirstIn(t, U("*\uFF0A")) Then t = TrimX(Mid$(t, 2))
                sched.Add "* " & t
            End If
        Next
    Next
End Sub

' ===========================================================================
' Schedule lines -> items (port of parseSched)
'   item = Array(time, mains As Collection, subs As Collection of Array(kind, text))
' ===========================================================================
Public Function IsBullet(ByVal t As String) As Boolean
    IsBullet = FirstIn(t, U("-\u2013\u25CB\u25CF\u25E6\u00B7\u2022\u25B6\u25B7\u25A1\u25A0"))
End Function

Public Function ParseSched(ByVal lines As Collection) As Variant
    Dim items As New Collection, refs As New Collection, hasLast As Boolean
    Dim curMains As Collection, curSubs As Collection, lastSub As Variant
    Dim l As Variant, raw As String, t As String, k As Long, pre As String, body As String
    Dim sp As String, rest As String, after As String, s As String
    For Each l In lines
        raw = CStr(l)
        If InStr(raw, vbLf) > 0 Then raw = Replace(raw, vbLf, " ")
        t = TrimX(raw)
        If t = "" Then GoTo nextLine
        If Left$(t, 1) = U("\u203B") Then
            refs.Add StripLead(t, U("\u203B"))
            GoTo nextLine
        End If
        If FirstIn(t, U("*\uFF0A")) Then
            s = StripLead(t, U("*\uFF0A"))
            If Not hasLast Then
                Set curMains = New Collection
                Set curSubs = New Collection
                items.Add Array("", curMains, curSubs)
                hasLast = True
            End If
            curSubs.Add Array("star", s)
            GoTo nextLine
        End If
        k = InStr(t, vbTab)
        If k > 1 And k <= 17 And Not IsBullet(t) Then
            pre = TrimX(Left$(t, k - 1))
            If SplitTime(pre, sp, rest) And rest = "" Then
                Set curMains = New Collection
                Set curSubs = New Collection
                items.Add Array(sp, curMains, curSubs)
            Else
                Set curMains = New Collection
                Set curSubs = New Collection
                items.Add Array(pre, curMains, curSubs)
            End If
            body = TrimX(TabsToSpace(Mid$(t, k + 1)))
            If body <> "" Then curMains.Add body
            hasLast = True
            GoTo nextLine
        End If
        t = TabsToSpace(t)
        If IsBullet(t) Then
            after = TrimX(Mid$(t, 2))
            If SplitTime(after, sp, rest) Then
                Set curMains = New Collection
                Set curSubs = New Collection
                items.Add Array(sp, curMains, curSubs)
                If rest <> "" Then curMains.Add rest
                hasLast = True
            ElseIf hasLast And FirstIn(t, U("-\u2013\u00B7\u2022")) Then
                curSubs.Add Array("dash", after)
            Else
                Set curMains = New Collection
                Set curSubs = New Collection
                items.Add Array("", curMains, curSubs)
                curMains.Add after
                hasLast = True
            End If
            GoTo nextLine
        End If
        If SplitTime(t, sp, rest) Then
            Set curMains = New Collection
            Set curSubs = New Collection
            items.Add Array(sp, curMains, curSubs)
            If rest <> "" Then curMains.Add rest
            hasLast = True
            GoTo nextLine
        End If
        If hasLast And (Left$(raw, 1) = " " Or Left$(raw, 1) = U("\u3000")) Then
            If curSubs.Count > 0 Then
                lastSub = curSubs(curSubs.Count)
                curSubs.Remove curSubs.Count
                curSubs.Add Array(lastSub(0), lastSub(1) & vbLf & t)
            Else
                curMains.Add t
            End If
            GoTo nextLine
        End If
        Set curMains = New Collection
        Set curSubs = New Collection
        items.Add Array("", curMains, curSubs)
        curMains.Add t
        hasLast = True
nextLine:
    Next
    ParseSched = Array(items, refs)
End Function

' ===========================================================================
' Rich text pieces
' ===========================================================================
'
Private Sub RBMainLine(ByVal s As String, Optional ByVal size As Double = 0)
    Dim p As Long, place As String, m1 As String
    p = InStrRev(s, "@")
    If p > 0 Then
        place = TrimX(Mid$(s, p + 1))
        If place <> "" Then
            If p = 1 Then
                RBAdd "(", , , , size
                RBInline place, , size
                RBAdd ")", , , , size
                Exit Sub
            ElseIf IsWs(Mid$(s, p - 1, 1)) Then
                m1 = RTrimX(Left$(s, p - 1))
                RBInline m1, , size
                If m1 <> "" Then RBAdd " ", , , , size
                RBAdd "(", , , , size
                RBInline place, , size
                RBAdd ")", , , , size
                Exit Sub
            End If
        End If
    End If
    RBInline s, , size
End Sub

' cell text of a bottom table (port of xtCellHTML)
Private Sub RBTableCell(ByVal txt As String, ByVal fs As Double)
    Dim l As Variant, t As String, n As Long, sp As String, rest As String, sfs As Double
    sfs = R5(MaxD(MaxD(fs - 1.5, 0.75 * fs), 6))
    For Each l In SplitLines(TabsToSpace(txt))
        t = TrimX(CStr(l))
        If t <> "" Then
            If n > 0 Then RBAdd vbLf
            n = n + 1
            If Left$(t, 1) = U("\u203B") Then
                RBInline t
            ElseIf FirstIn(t, U("*\uFF0A")) Then
                RBAdd "* ", , , , sfs
                RBInline StripLead(t, U("*\uFF0A")), , sfs
            ElseIf SplitTime(t, sp, rest) Then
                RBAdd sp, True
                If rest <> "" Then
                    RBAdd " "
                    RBMainLine rest
                End If
            Else
                RBMainLine t
            End If
        End If
    Next
End Sub

End Function

' header block (title, secret box, sub/date line). returns height incl. the space below
Private Function LayHead(ByVal doRender As Boolean) As Double
    Dim tH As Double, secW As Double, secH As Double, tx As Double, y As Double, mfs As Double, mH As Double, idx As Long
    Dim dTxt As String
    gTrace = "LayHead start"
    mfs = MetaFs()
    gTrace = "LayHead mfs"
    If gShowSecret Then
        secW = MeasureW(gSecretText, gFont, 9.5, True) + 3.4 * PT_MM
        secH = MeasureH(gSecretText, "", secW, gFont, 9.5, True) + 1 * PT_MM
        If doRender Then
            idx = ElText(gW - secW, 0, secW, secH, gSecretText, "", 9.5, True, XL_CENTER, XL_CENTER, COL_RED, False)
            ElBorderAll idx, XL_CONT, XL_MEDIUM, COL_RED
        End If
        tx = secW + 2 * PT_MM
    End If
    If gTitle <> "" Then
        tH = MeasureH(gTitle, "", gW - 2 * tx, gFont, TitleFs(), True)
        If doRender Then ElText tx, 0, gW - 2 * tx, tH, gTitle, "", TitleFs(), True, XL_CENTER, XL_CENTER
        y = tH
    Else
        y = secH
    End If
    y = y + 1.2 * PT_MM
    dTxt = FmtDateK(gDate)
    mH = MaxD(MeasureH(IIf(gSub = "", U("\uAC00"), gSub), "", gW * 0.62, gFont, mfs), MeasureH(dTxt, "", gW * 0.38, gFont, mfs, True))
    If doRender Then
        If gSub <> "" Then ElText 0, y, gW * 0.62, mH, gSub, "", mfs, False, XL_LEFT, XL_BOTTOM
        ElText gW * 0.62, y, gW * 0.38, mH, dTxt, "", mfs, True, XL_RIGHT, XL_BOTTOM
    End If
    y = y + mH + 1.6 * PT_MM
    If doRender Then
        idx = ElBox(0, 0, gW, y)
        ElBorder idx, 1, XL_CONT, XL_MEDIUM, 0
    End If
    LayHead = y + 3 * PT_MM
End Function

Private Function LayFoot(ByVal y As Double, ByVal doRender As Boolean) As Double
    If gFoot = "" Then Exit Function
    Dim h As Double
    h = MeasureH(gFoot, "", gW, gFont, MetaFs())
    If doRender Then ElText 0, y + 2.2 * PT_MM, gW, h, gFoot, "", MetaFs()
    LayFoot = h + 2.2 * PT_MM
End Function

' width of the time column in box i
Private Function TimeColW(ByVal i As Long) As Double
    Dim it As Variant, p As Variant, w As Double
    For Each it In exItems(i)
        If it(0) <> "" Then
            For Each p In Split(TimePieces(it(0)), vbLf)
                w = MaxD(w, MeasureW(CStr(p), gFont, gFS, True))
            Next
        End If
    Next
    TimeColW = w
End Function

Private Function TimePieces(ByVal t As String) As String
    TimePieces = Replace(t, "~", "~" & vbLf)
End Function

' one executive box. returns natural height; when doRender, draws it with height h
Private Function LayBox(ByVal i As Long, ByVal x As Double, ByVal y As Double, ByVal w As Double, ByVal h As Double, ByVal doRender As Boolean) As Double
    Dim fs As Double, pv As Double, ph As Double, rp As Double
    fs = gFS: pv = PadV(): ph = PadH(): rp = RowPad()
    Dim hfs As Double, headTxt As String, headH As Double, idx As Long
    hfs = R5(fs * 1.06)
    headTxt = TrimX(exPos(i) & " " & exName(i))
    headH = MeasureH(IIf(headTxt = "", U("\uAC00"), headTxt), "", w - 2 * ph, gFont, hfs, True) + 1.4 * pv
    If doRender Then
        idx = ElText(x, y, w, headH, headTxt, "", hfs, True, XL_CENTER, XL_CENTER)
        ElFill idx, COL_HEAD
        ElBorder idx, 1, XL_CONT, XL_MEDIUM, COL_LINE
        idx = ElBox(x, y, w, h)
        ElBorderAll idx, XL_CONT, XL_THIN, COL_LINE
    End If
    Dim x0 As Double, bw As Double, yy As Double, tw As Double, cx As Double, cw As Double
    x0 = x + ph - CELLPAD
    bw = w - 2 * ph + 2 * CELLPAD
    yy = y + headH + 0.5 * pv
    tw = TimeColW(i)
    If tw > 0 Then
        tw = tw + MIN_COL
        cx = x0 + tw + 0.15 * fs
    Else
        cx = x0
    End If
    cw = x0 + bw - cx
    Dim k As Long, it As Variant, m As Long, itemTop As Double, y1 As Double, cy As Double
    Dim tTxt As String, tH As Double, mTxt As String, mRuns As String, mMark As Boolean, mH As Double
    Dim sb As Variant, sfs As Double, mk As String, mkW As Double, sTxt As String, sRuns As String, sMark As Boolean, sH As Double
    Dim l As Variant, n As Long, mains As Collection, subs As Collection, itemH As Double, nh As Double
    Dim refH As Double, rfs As Double, rmW As Double, r As Variant, rTop As Double, rH() As Double, rT() As String, rR() As String, rM() As Boolean
    k = 0
    For Each it In exItems(i)
        k = k + 1
        itemTop = yy
        y1 = yy + rp
        tTxt = TimePieces(it(0))
        Set mains = it(1)
        Set subs = it(2)
        tH = 0
        If tTxt <> "" Then tH = MeasureH(tTxt, "", tw, gFont, fs, True)
        RBReset
        For m = 1 To mains.Count
            If m > 1 Then RBAdd vbLf
            RBMainLine mains(m)
        Next
        mTxt = RBText(): mRuns = RBRuns(): mMark = rbMark
        mH = MeasureH(mTxt, mRuns, cw, gFont, fs)
        If doRender And mTxt <> "" Then
            idx = ElText(cx, y1, cw, mH, mTxt, mRuns, fs)
            If mMark Then ElFill idx, COL_MARK
        End If
        cy = y1 + mH
        For Each sb In subs
            If sb(0) = "star" Then
                sfs = R5(MaxD(MaxD(fs - 2, 0.75 * fs), 6))
                mk = "*"
                cy = cy + 0.06 * fs
            Else
                sfs = fs
                mk = "-"
                cy = cy + 0.02 * fs
            End If
            mkW = MeasureW(mk, gFont, sfs) + 0.1 * sfs
            RBReset
            n = 0
            For Each l In Split(sb(1), vbLf)
                If n > 0 Then RBAdd vbLf
                n = n + 1
                RBInline CStr(l)
            Next
            sTxt = RBText(): sRuns = RBRuns(): sMark = rbMark
            sH = MeasureH(sTxt, sRuns, cw - mkW, gFont, sfs)
            If doRender Then
                ElText cx, cy, mkW, sH, mk, "", sfs
                idx = ElText(cx + mkW, cy, cw - mkW, sH, sTxt, sRuns, sfs)
                If sMark Then ElFill idx, COL_MARK
            End If
            cy = cy + sH
        Next
        itemH = MaxD(tH, cy - y1) + 2 * rp
        If doRender Then
            If tTxt <> "" Then ElText x0, y1, tw, tH, tTxt, "", fs, True
            If k > 1 Then
                idx = ElBox(x0, itemTop, bw, itemH)
                ElBorder idx, 0, XL_DOT, XL_HAIR, COL_DOT
            End If
        End If
        yy = itemTop + itemH
    Next
    If k = 0 And gEmptyText <> "" Then
        nh = MeasureH(gEmptyText, "", bw, gFont, fs)
        If doRender Then ElText x0, yy + 0.5 * fs, bw, nh, gEmptyText, "", fs, False, XL_CENTER, XL_TOP, COL_GRAY
        yy = yy + nh + fs
    End If
    yy = yy + 0.5 * pv
    '
    If exRefs(i).Count > 0 Then
        rfs = R5(0.93 * fs)
        rmW = MeasureW(U("\u203B"), gFont, rfs)
        ReDim rH(1 To exRefs(i).Count): ReDim rT(1 To exRefs(i).Count): ReDim rR(1 To exRefs(i).Count): ReDim rM(1 To exRefs(i).Count)
        refH = 0.8 * pv + 0.9 * pv
        n = 0
        For Each r In exRefs(i)
            n = n + 1
            RBReset
            RBInline CStr(r)
            rT(n) = RBText(): rR(n) = RBRuns(): rM(n) = rbMark
            rH(n) = MeasureH(rT(n), rR(n), bw - rmW, gFont, rfs)
            refH = refH + rH(n)
            If n > 1 Then refH = refH + 0.08 * fs
        Next
        If doRender Then
            rTop = y + h - refH
            idx = ElBox(x0, rTop, bw, refH)
            ElBorder idx, 0, XL_DASH, XL_HAIR, COL_DASH
            cy = rTop + 0.8 * pv
            For n = 1 To exRefs(i).Count
                If n > 1 Then cy = cy + 0.08 * fs
                ElText x0, cy, rmW, rH(n), U("\u203B"), "", rfs
                idx = ElText(x0 + rmW, cy, bw - rmW, rH(n), rT(n), rR(n), rfs)
                If rM(n) Then ElFill idx, COL_MARK
                cy = cy + rH(n)
            Next
        End If
    End If
    LayBox = (yy - y) + refH
End Function

' bottom tables. returns total height
Private Function LayTables(ByVal y0 As Double, ByVal doRender As Boolean) As Double
    Dim tf As Double, pvT As Double, phT As Double, lw As Double, k As Long, j As Long, n As Long, y As Double
    Dim colW As Double, heads As Boolean, hRow As Double, cRow As Double, hh As Double, idx As Long
    tf = R5(0.95 * gFS)
    pvT = (0.22 + 0.3 * gU) * 0.55 * tf
    phT = (0.4 + 0.2 * gU) * 0.8 * tf
    For k = 1 To 4
        lw = MaxD(lw, MeasureW(IIf(tbLabel(k) = "", U("\uAC00"), tbLabel(k)), gFont, tf, True))
    Next
    lw = Clamp(lw + 0.6 * tf, 3 * tf, 10 * tf)
    y = y0
    Dim cT(1 To 5) As String, cR(1 To 5) As String, cM(1 To 5) As Boolean, cH As Double
    For k = 1 To 4
        If k > 1 Then y = y + gGap
        n = TblN(k)
        colW = (gW - lw) / n
        heads = False
        If TblHasHeads(k) Then
            For j = 1 To n
                If tbHead(k, j) <> "" Then heads = True
            Next
        End If
        hRow = 0
        If heads Then
            For j = 1 To n
                hRow = MaxD(hRow, MeasureH(IIf(tbHead(k, j) = "", U("\uAC00"), tbHead(k, j)), "", colW - 2 * phT, gFont, tf, True))
            Next
            hRow = hRow + 2 * pvT
        End If
        cRow = MeasureH(U("\uAC00"), "", colW - 2 * phT, gFont, tf)
        For j = 1 To n
            RBReset
            RBTableCell tbText(k, j), tf
            cT(j) = RBText(): cR(j) = RBRuns(): cM(j) = rbMark
            cRow = MaxD(cRow, MeasureH(cT(j), cR(j), colW - 2 * phT + 2 * CELLPAD, gFont, tf))
        Next
        cRow = cRow + 2 * pvT
        If doRender Then
            idx = ElText(0, y, lw, hRow + cRow, tbLabel(k), "", tf, True, XL_CENTER, XL_CENTER, 0, False)
            ElFill idx, COL_HEAD
            ElBorderAll idx, XL_CONT, XL_THIN, COL_LINE
            For j = 1 To n
                If heads Then
                    idx = ElText(lw + (j - 1) * colW, y, colW, hRow, tbHead(k, j), "", tf, True, XL_CENTER, XL_CENTER)
                    ElFill idx, COL_HEAD2
                    ElBorderAll idx, XL_CONT, XL_THIN, COL_LINE
                End If
                idx = ElBox(lw + (j - 1) * colW, y + hRow, colW, cRow)
                ElBorderAll idx, XL_CONT, XL_THIN, COL_LINE
                If cT(j) <> "" Then
                    idx = ElText(lw + (j - 1) * colW + phT - CELLPAD, y + hRow + pvT, colW - 2 * phT + 2 * CELLPAD, cRow - 2 * pvT, cT(j), cR(j), tf)
                    If cM(j) Then ElFill idx, COL_MARK
                End If
            Next
        End If
        y = y + hRow + cRow
    Next
    LayTables = y - y0
End Function

' box widths for seat i
Private Sub SeatGeom(ByVal i As Long, ByRef x As Double, ByRef w As Double)
    Dim a As Double
    Select Case i
        Case 1, 2, 3
            a = gW - 2 * gGap
            x = 0
            If i >= 2 Then x = gC1(1) * a + gGap
            If i = 3 Then x = x + gC1(2) * a + gGap
            w = gC1(i) * a
        Case 4
            x = 0: w = gC2(1) * (gW - gGap)
        Case 5, 6
            x = gC2(1) * (gW - gGap) + gGap: w = gC2(2) * (gW - gGap)
        Case 7
            x = 0: w = gC3(1) * (gW - gGap)
        Case 8
            x = gC3(1) * (gW - gGap) + gGap: w = gC3(2) * (gW - gGap)
    End Select
End Sub

Private Sub FitEval(ByVal fs As Double, ByVal lo As Double, ByVal hi As Double, ByRef ok As Boolean, ByRef need As Double, ByRef avail As Double, _
        ByRef req() As Double, ByRef nat() As Double, ByRef headH As Double, ByRef tabH As Double, ByRef footH As Double)
    Dim i As Long, x As Double, w As Double
    gTrace = "FitEval start fs=" & fs
    gFS = fs
    gTrace = "FitEval gFS"
    If hi > lo Then
        gU = Clamp((fs - lo) / (hi - lo), 0, 1)
    Else
        gU = 1
    End If
    gTrace = "FitEval head"
    headH = LayHead(False)
    gTrace = "FitEval foot"
    footH = LayFoot(0, False)
    tabH = 0
    If gShowTables Then tabH = LayTables(0, False)
    avail = gH - headH - footH
    If gShowTables Then avail = avail - tabH - gGap
    For i = 1 To NX
        SeatGeom i, x, w
        nat(i) = LayBox(i, x, 0, w, 0, False)
    Next
    req(1) = MaxD(MaxD(nat(1), nat(2)), nat(3))
    If gSplit < 0 Then
        req(2) = MaxD(nat(4), nat(5) + gGap + nat(6))
    Else
        req(2) = MaxD(MaxD(nat(4), nat(5) / gSplit + gGap), nat(6) / (1 - gSplit) + gGap)
    End If
    req(3) = MaxD(nat(7), nat(8))
    Dim inner As Double, t As Double, k As Long
    inner = avail - 2 * gGap
    If Not gRowsSet Then
        need = req(1) + req(2) + req(3) + 2 * gGap
        ok = (need <= avail - 1.5)
    Else
        ok = True
        need = 2 * gGap
        For k = 1 To 3
            t = gRows(k) * (inner - 1)
            If req(k) > t Then ok = False
            need = need + MaxD(req(k), t)
        Next
    End If
End Sub

Public Sub DoBuild()
    gStep = "settings"
    LoadSettings
    gStep = "execs"
    LoadExecs
    gStep = "sync names"
    SyncExecNames
    gStep = "tables"
    LoadTables
    gStep = "measure init"
    ' row AutoFit (used to measure text) needs screen updating on in some hosts;
    ' only the hidden scratch sheet changes while measuring
    SetScreen True
    MsInit SH_MS
    gStep = "fit"
    Dim lo As Double, hi As Double, fs As Double, ok As Boolean, need As Double, avail As Double
    Dim req() As Double, nat() As Double, headH As Double, tabH As Double, footH As Double, h() As Double
    Dim a As Long, b As Long, md As Long
    ReDim req(1 To 3): ReDim nat(1 To 8): ReDim h(1 To 3)
    gTrace = "fit lo/hi"
    lo = MinD(gMinF, gMaxF): hi = MaxD(gMinF, gMaxF)
    lo = R5(lo): hi = R5(hi)
    gTrace = "fit call"
    FitEval hi, lo, hi, ok, need, avail, req, nat, headH, tabH, footH
    If ok Then
        fs = hi
    Else
        FitEval lo, lo, hi, ok, need, avail, req, nat, headH, tabH, footH
        fs = lo
        If ok Then
            a = CLng(lo * 2): b = CLng(hi * 2)
            Do While b - a > 1
                md = (a + b) \ 2
                FitEval md / 2, lo, hi, ok, need, avail, req, nat, headH, tabH, footH
                If ok Then
                    a = md
                Else
                    b = md
                End If
            Loop
            fs = a / 2
            FitEval fs, lo, hi, ok, need, avail, req, nat, headH, tabH, footH
        End If
    End If
    ' heights of the three rows
    Dim inner As Double, extra As Double, k As Long, in2 As Double, ex2 As Double, hE As Double, hF As Double
    inner = avail - 2 * gGap
    If ok Then
        If Not gRowsSet Then
            extra = MaxD(0, inner - 1 - req(1) - req(2) - req(3))
            h(1) = req(1) + extra / 4: h(2) = req(2) + extra / 2: h(3) = req(3) + extra / 4
        Else
            For k = 1 To 3
                h(k) = gRows(k) * (inner - 1)
            Next
        End If
    Else
        For k = 1 To 3
            h(k) = req(k)
            If gRowsSet Then h(k) = MaxD(req(k), gRows(k) * (inner - 1))
        Next
    End If
    in2 = h(2) - gGap
    If gSplit < 0 Then
        ex2 = MaxD(0, in2 - nat(5) - nat(6))
        hE = nat(5) + ex2 / 2: hF = nat(6) + ex2 / 2
    Else
        hE = MaxD(gSplit * in2, nat(5)): hF = MaxD(in2 - gSplit * in2, nat(6))
    End If
    h(2) = MaxD(MaxD(h(2), hE + gGap + hF), nat(4))
    ' draw
    gStep = "render"
    SetScreen False
    ElReset
    Dim y As Double, x As Double, w As Double, i As Long
    y = LayHead(True)
    For i = 1 To 3
        SeatGeom i, x, w
        LayBox i, x, y, w, h(1), True
    Next
    y = y + h(1) + gGap
    SeatGeom 4, x, w: LayBox 4, x, y, w, h(2), True
    SeatGeom 5, x, w: LayBox 5, x, y, w, hE, True
    SeatGeom 6, x, w: LayBox 6, x, y + hE + gGap, w, h(2) - hE - gGap, True
    y = y + h(2) + gGap
    SeatGeom 7, x, w: LayBox 7, x, y, w, h(3), True
    SeatGeom 8, x, w: LayBox 8, x, y, w, h(3), True
    y = y + h(3)
    If gShowTables Then
        y = y + gGap
        y = y + LayTables(y, True)
    End If
    If ok And gFoot <> "" Then y = gH - footH
    LayFoot y, True
    Dim ws As Object, lastR As Long, lastC As Long
    Set ws = HostBook().Worksheets(SH_A4)
    gStep = "realize"
    Realize ws, gFont, lastR, lastC
    gStep = "page setup"
    PageSetupA4 ws, gLandscape, gMargin, lastR, lastC
    gStep = "status"
    ' status
    Dim st As String
    If ok Then
        st = U("A4 \uD55C \uC7A5 \uC790\uB3D9 \uB9DE\uCDA4 \u00B7 \uAE00\uC790 ") & NumStr(fs) & "pt"
        If Not gRowsSet And gSplit < 0 And fs >= hi Then st = st & U(" \u00B7 \uCD5C\uB300 \uAE00\uC790\uD06C\uAE30 \u00B7 \uC5EC\uC720 ") & Format$((avail - need) / PT_MM, "0") & U("mm (\uCE78\uC744 \uB298\uB824 \uCC44\uC6C0)")
        If gRowsSet Or gSplit >= 0 Then st = st & U(" \u00B7 \uCE78 \uB192\uC774 \uC9C1\uC811 \uC870\uC815")
    Else
        st = U("A4 \uD55C \uC7A5\uC744 ") & Format$(-Int(-(need - avail) / PT_MM), "0") & U("mm \uB118\uC2B5\uB2C8\uB2E4 (\uAE00\uC790 ") & NumStr(fs) & "pt). "
        If gRowsSet Or gSplit >= 0 Then
            st = st & U("\uB192\uC774\uB97C \uC9C1\uC811 \uC815\uD55C \uCE78\uC5D0 \uB0B4\uC6A9\uC774 \uB118\uCE69\uB2C8\uB2E4. \uC11C\uC2DD \uC2DC\uD2B8\uC5D0\uC11C [\uB192\uC774\uB9CC \uC790\uB3D9\uC73C\uB85C]\uB97C \uB204\uB974\uC138\uC694.")
        Else
            st = st & U("\uB0B4\uC6A9\uC744 \uC904\uC774\uAC70\uB098 \uC11C\uC2DD \uC2DC\uD2B8\uC5D0\uC11C \uCD5C\uC18C \uAE00\uC790\uD06C\uAE30\u00B7\uC5EC\uBC31\uC744 \uC904\uC5EC \uBCF4\uC138\uC694. \uC9C0\uAE08 \uC778\uC1C4\uD558\uBA74 \uD55C \uC7A5\uC5D0 \uB9DE\uAC8C \uC904\uC5EC \uCC0D\uD799\uB2C8\uB2E4.")
        End If
    End If
    ShowStatus st, ok
    gStep = "sizes"
    UpdateSizes h, hE, hF
    gStep = "map"
    UpdateMap
    gStep = "a4 buttons"
    PlaceA4Buttons ws, lastC
    gStep = "done"
End Sub

Private Sub ShowStatus(ByVal st As String, ByVal ok As Boolean)
    Dim r As Object
    Set r = NameRange("d_status")
    If Not r Is Nothing Then
        r.Cells(1, 1).Value = IIf(ok, U("\uB9DE\uCDA4 \uC644\uB8CC: "), U("\uB118\uCE68: ")) & st
        r.Cells(1, 1).Font.Color = IIf(ok, COL_OK, COL_BAD)
    End If
    On Error Resume Next
    Dim ws As Object, shp As Object
    Set ws = HostBook().Worksheets(SH_A4)
    Set shp = ws.Shapes("a4status")
    If Not shp Is Nothing Then
        shp.TextFrame.Characters.Text = IIf(ok, U("\uB9DE\uCDA4 \uC644\uB8CC: "), U("\uB118\uCE68: ")) & st
        shp.TextFrame.Characters.Font.Color = IIf(ok, COL_OK, COL_BAD)
    End If
    On Error GoTo 0
End Sub

Private Sub PlaceA4Buttons(ByVal ws As Object, ByVal lastC As Long)
    Dim x As Double, shp As Object, y As Double
    On Error Resume Next
    x = ws.Cells(1, lastC + 1).Left + 18
    y = 6
    For Each shp In ws.Shapes
        If shp.Name = "a4status" Then
            shp.Left = x
            shp.Top = 6
        Else
            shp.Left = x
            shp.Top = 80 + y
            y = y + shp.Height + 6
        End If
    Next
    On Error GoTo 0
End Sub

Private Sub UpdateSizes(h() As Double, ByVal hE As Double, ByVal hF As Double)
    Dim r As Object, a As Double, b As Double
    Set r = NameRange("s_sizes")
    If r Is Nothing Then Exit Sub
    a = (gW - 2 * gGap) / PT_MM
    b = (gW - gGap) / PT_MM
    r.Cells(1, 1).Value = U("\uC717\uC904 ") & Format$(gC1(1) * a, "0") & U(" \u00B7 ") & Format$(gC1(2) * a, "0") & U(" \u00B7 ") & Format$(gC1(3) * a, "0") & _
        U("mm / \uAC00\uC6B4\uB370 \uC904 ") & Format$(gC2(1) * b, "0") & U(" \u00B7 ") & Format$(gC2(2) * b, "0") & _
        U("mm / \uC544\uB7AB\uC904 ") & Format$(gC3(1) * b, "0") & U(" \u00B7 ") & Format$(gC3(2) * b, "0") & "mm" & vbLf & _
        U("\uB192\uC774: \uC717\uC904 ") & Format$(h(1) / PT_MM, "0") & U(" \u00B7 \uAC00\uC6B4\uB370 \uC904 ") & Format$(h(2) / PT_MM, "0") & U(" (\uC624\uB978\uCABD \uC704 ") & Format$(hE / PT_MM, "0") & _

Private Sub UpdateMap()
    Dim r As Object, ws As Object, i As Long, rr As Long, c1 As Long, c2 As Long, r2 As Long, rg As Object, txt As String
    Set r = NameRange("d_map")
    If r Is Nothing Then Exit Sub
    Set ws = r.Worksheet
    Application.DisplayAlerts = False
    For i = 1 To NX
        Select Case i
            Case 1: rr = 0: r2 = 0: c1 = 0: c2 = 3
            Case 2: rr = 0: r2 = 0: c1 = 4: c2 = 7
            Case 3: rr = 0: r2 = 0: c1 = 8: c2 = 11
            Case 4: rr = 1: r2 = 2: c1 = 0: c2 = 7
            Case 5: rr = 1: r2 = 1: c1 = 8: c2 = 11
            Case 6: rr = 2: r2 = 2: c1 = 8: c2 = 11
            Case 7: rr = 3: r2 = 3: c1 = 0: c2 = 5
            Case 8: rr = 3: r2 = 3: c1 = 6: c2 = 11
        End Select
        Set rg = ws.Range(r.Offset(rr, c1), r.Offset(r2, c2))
        On Error Resume Next
        rg.Hyperlinks.Delete
        On Error GoTo 0
        If rg.Cells.Count > 1 And Not rg.MergeCells Then rg.Merge
        txt = i & ". " & TrimX(exPos(i) & " " & exName(i))
        txt = txt & vbLf & U("\uC77C\uC815 ") & exItems(i).Count & U("\uAC74")
        If exRefs(i).Count > 0 Then txt = txt & U(" \u00B7 \u203B ") & exRefs(i).Count & U("\uC904")
        rg.Cells(1, 1).Value = txt
        rg.HorizontalAlignment = XL_CENTER
        rg.VerticalAlignment = XL_CENTER
        rg.WrapText = True
        rg.Font.Name = U("\uB9D1\uC740 \uACE0\uB515")
        rg.Font.Size = 9
        rg.Font.Color = 0
        rg.Interior.Color = IIf(exItems(i).Count = 0, COL_SOFT, 16777215)
        rg.Borders.LineStyle = XL_CONT
        rg.Borders.Color = COL_INLINE
        AddLink ws, rg.Cells(1, 1), "#'" & exWs(i).Name & "'!B4", 0
        If Len(txt) > 0 Then rg.Cells(1, 1).Characters(1, InStr(txt, vbLf) - 1).Font.Bold = True
    Next
    Application.DisplayAlerts = True
    ' bottom tables line
    Dim t As Object, filled As Long, k As Long, j As Long
    Set t = r.Offset(4, 0)
    For k = 1 To 4
        For j = 1 To TblN(k)
            If TrimX(tbText(k, j)) <> "" Then
                filled = filled + 1
                Exit For
            End If
        Next
    Next
    t.Value = U("\uB9E8 \uBC11 \uD45C: \uC778\uC811\uD68C\uC0AC\uD65C\uB3D9 \u00B7 \uD734\uAC00 \u00B7 \uCD9C\uC7A5/\uBC29\uBB38 \u00B7 \uADFC\uBB34 \u2014 ") & filled & U("/4\uAC1C \uC785\uB825") & IIf(gShowTables, "", U(" (\uC11C\uC2DD\uC5D0\uC11C \uAEBC \uB460)"))
    AddLink ws, t, "#'" & SH_TBL & "'!D4", COL_ACCENT
End Sub

Private Sub LoadTables()
    Dim r As Object, k As Long, j As Long, row As Long
    Set r = NameRange("t_top")
    For k = 1 To 4
        For j = 1 To 5
            tbHead(k, j) = "": tbText(k, j) = ""
        Next
        tbLabel(k) = Choose(k, U("\uC778\uC811\uD68C\uC0AC\uD65C\uB3D9"), U("\uD734\uAC00"), U("\uCD9C\uC7A5/\uBC29\uBB38"), U("\uADFC\uBB34"))
        If Not r Is Nothing Then
            row = TblRow0(k)
            tbLabel(k) = CellText(r.Offset(row, 0).Value)
            For j = 1 To TblN(k)
                If TblHasHeads(k) Then tbHead(k, j) = CellText(r.Offset(row + j - 1, 2).Value)
                tbText(k, j) = CellText(r.Offset(row + j - 1, 3).Value)
            Next
        End If
    Next
End Sub
