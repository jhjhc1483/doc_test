' ===========================================================================
' 한셀 매크로 점검 — 한셀 매크로 편집기에서 어떤 기능이 되는지 확인합니다.
'
' 쓰는 법
'   1. 한셀에서 새 문서를 엽니다.
'   2. 매크로 편집기(도구 → 매크로 → 스크립트/VBA 편집기)에서 모듈을 하나 만들고
'      이 파일 내용을 전부 붙여넣습니다.
'   3. HancellCheck 를 실행합니다. ("점검을 시작합니다" 창이 떠야 합니다)
'   4. 끝나면 '점검' 시트가 생깁니다. 실패한 줄을 알려 주시거나 그 시트를 캡처해 주세요.
'   5. (선택) 점검 시트에 생긴 [단추 점검] 단추를 눌러 보고, HancellFileTest 도 실행해 보세요.
'
' 일부러 한셀과 엑셀이 모두 읽는 문법만 썼습니다 (변수 형식·$ 함수·IIf 없음).
' ===========================================================================

Sub HancellCheck()
    Dim wb, ws, tmp, r, okN, fails, v, t, n, b, d, i
    MsgBox "점검을 시작합니다"
    okN = 0
    fails = ""
    r = 1
    Set ws = Nothing
    On Error Resume Next

    Err.Clear
    t = Application.Name & " " & Application.Version
    RecLine ws, r, "01 Application.Name/Version", Err.Number, Err.Description, t, okN, fails

    Err.Clear
    Set wb = ActiveWorkbook
    t = wb.Name
    RecLine ws, r, "02 ActiveWorkbook", Err.Number, Err.Description, t, okN, fails

    Err.Clear
    t = ThisWorkbook.Name
    RecLine ws, r, "03 ThisWorkbook", Err.Number, Err.Description, t, okN, fails

    Err.Clear
    t = wb.Worksheets.Count
    RecLine ws, r, "04 Worksheets.Count", Err.Number, Err.Description, t, okN, fails

    Err.Clear
    Set tmp = Nothing
    Set tmp = wb.Worksheets.Add(, wb.Worksheets(wb.Worksheets.Count))
    RecLine ws, r, "05 Worksheets.Add(, 뒤)", Err.Number, Err.Description, "", okN, fails
    If tmp Is Nothing Then
        Err.Clear
        Set tmp = wb.Worksheets.Add()
        RecLine ws, r, "05b Worksheets.Add()", Err.Number, Err.Description, "", okN, fails
    End If

    Err.Clear
    n = "점검"
    For i = 2 To 9
        Set d = Nothing
        Set d = wb.Worksheets(n)
        If d Is Nothing Then Exit For
        n = "점검" & i
    Next
    Err.Clear
    tmp.Name = n
    RecLine ws, r, "06 시트 이름 바꾸기", Err.Number, Err.Description, tmp.Name, okN, fails
    Set ws = tmp
    ws.Cells(1, 1).Value = "항목"
    ws.Cells(1, 2).Value = "결과"
    ws.Columns(1).ColumnWidth = 34
    ws.Columns(2).ColumnWidth = 70

    Err.Clear
    ws.Range("D1").Value = "가나다"
    ws.Cells(2, 4).Value = 123
    t = ws.Range("D1").Value & "/" & ws.Cells(2, 4).Value
    RecLine ws, r, "07 Range/Cells 값 쓰기·읽기", Err.Number, Err.Description, t, okN, fails

    Err.Clear
    ws.Range("D3").Font.Bold = True
    ws.Range("D3").Font.Size = 14
    ws.Range("D3").Font.Name = "맑은 고딕"
    ws.Range("D3").Font.Color = RGB(196, 0, 0)
    ws.Range("D3").Value = "글꼴"
    RecLine ws, r, "08 Font (Bold/Size/Name/Color)", Err.Number, Err.Description, "", okN, fails

    Err.Clear
    ws.Range("D4").Interior.Color = RGB(255, 235, 90)
    RecLine ws, r, "09 Interior.Color (칸 채우기)", Err.Number, Err.Description, "", okN, fails

    Err.Clear
    ws.Range("D5").Borders(9).LineStyle = 1
    ws.Range("D5").Borders(9).Weight = -4138
    ws.Range("D5").Borders(9).Color = RGB(0, 0, 0)
    RecLine ws, r, "10 Borders(아래) 선", Err.Number, Err.Description, "", okN, fails

    Err.Clear
    ws.Range("D6:E6").Borders.LineStyle = 1
    ws.Range("D7").Borders(8).LineStyle = -4118
    RecLine ws, r, "11 Borders 전체·점선", Err.Number, Err.Description, "", okN, fails

    Err.Clear
    ws.Range("D10:F12").Merge
    t = ws.Range("D10").MergeCells
    RecLine ws, r, "12 Merge (병합)", Err.Number, Err.Description, "MergeCells=" & t, okN, fails

    Err.Clear
    ws.Columns(7).ColumnWidth = 20
    t = ws.Columns(7).Width
    ws.Columns(8).ColumnWidth = 0.5
    t = t & " / 0.5폭=" & ws.Columns(8).Width
    RecLine ws, r, "13 ColumnWidth → Width(pt)", Err.Number, Err.Description, "20폭=" & t, okN, fails

    Err.Clear
    ws.Rows(20).RowHeight = 3.75
    t = ws.Rows(20).RowHeight
    ws.Rows(21).RowHeight = 0.75
    t = t & " / " & ws.Rows(21).RowHeight
    RecLine ws, r, "14 RowHeight 작게", Err.Number, Err.Description, t, okN, fails

    Err.Clear
    ws.Columns(9).ColumnWidth = 10
    ws.Cells(22, 9).WrapText = True
    ws.Cells(22, 9).Value = "줄이 여러 번 바뀌도록 길게 쓴 글입니다. 자동 맞춤이 되면 줄 높이가 커집니다."
    ws.Rows(22).AutoFit
    t = ws.Rows(22).RowHeight
    RecLine ws, r, "15 Rows.AutoFit (글 높이 재기)", Err.Number, Err.Description, "높이=" & t & " (15보다 커야 정상)", okN, fails

    Err.Clear
    ws.Cells(23, 10).Value = "열 너비 자동 맞춤 시험 글"
    ws.Columns(10).AutoFit
    t = ws.Columns(10).Width
    RecLine ws, r, "16 Columns.AutoFit (글 폭 재기)", Err.Number, Err.Description, "폭=" & t, okN, fails

    Err.Clear
    ws.Range("D24").Value = "09:00 회의"
    ws.Range("D24").Characters(1, 5).Font.Bold = True
    ws.Range("D24").Characters(7, 2).Font.Color = RGB(196, 0, 0)
    RecLine ws, r, "17 Characters (칸 안 일부 굵게)", Err.Number, Err.Description, "D24 글자 일부가 굵게 보이면 정상", okN, fails

    Err.Clear
    ws.Range("D25").NumberFormat = "@"
    ws.Range("D25").Value = "09:00"
    t = TypeName(ws.Range("D25").Value) & " " & ws.Range("D25").Value
    RecLine ws, r, "18 NumberFormat=@ (텍스트)", Err.Number, Err.Description, t, okN, fails

    Err.Clear
    ws.Range("D26").HorizontalAlignment = -4108
    ws.Range("D26").VerticalAlignment = -4160
    RecLine ws, r, "19 정렬 (가운데·위)", Err.Number, Err.Description, "", okN, fails

    Err.Clear
    ws.PageSetup.PaperSize = 9
    ws.PageSetup.Orientation = 1
    ws.PageSetup.LeftMargin = 28
    ws.PageSetup.TopMargin = 28
    RecLine ws, r, "20 PageSetup 용지·여백", Err.Number, Err.Description, "", okN, fails

    Err.Clear
    ws.PageSetup.Zoom = False
    ws.PageSetup.FitToPagesWide = 1
    ws.PageSetup.FitToPagesTall = 1
    RecLine ws, r, "21 PageSetup 한 페이지에 맞춤", Err.Number, Err.Description, "", okN, fails

    Err.Clear
    ws.PageSetup.PrintArea = "$A$1:$F$40"
    ws.PageSetup.CenterHorizontally = True
    RecLine ws, r, "22 PageSetup 인쇄 영역", Err.Number, Err.Description, "", okN, fails

    Err.Clear
    ws.Hyperlinks.Add ws.Range("D27"), "http://www.hancom.com"
    RecLine ws, r, "23 Hyperlinks.Add (링크)", Err.Number, Err.Description, "", okN, fails

    Err.Clear
    ws.Range("D28").AddComment "마우스를 올리면 보이는 메모"
    RecLine ws, r, "24 AddComment (메모)", Err.Number, Err.Description, "", okN, fails

    Err.Clear
    wb.Names.Add "hc_test", "='" & ws.Name & "'!$D$1"
    t = wb.Names("hc_test").RefersToRange.Value
    RecLine ws, r, "25 Names (이름 정의)", Err.Number, Err.Description, t, okN, fails

    Err.Clear
    ws.Range("D29").Validation.Add 3, 1, 1, "예,아니오"
    RecLine ws, r, "26 Validation 목록", Err.Number, Err.Description, "", okN, fails

    Err.Clear
    Set b = Nothing
    Set b = ws.Buttons.Add(420, 60, 110, 26)
    b.Caption = "단추 점검"
    b.OnAction = "HancellCheckClick"

    Err.Clear
    Set b = Nothing

    Err.Clear
    Application.ScreenUpdating = False
    Application.ScreenUpdating = True
    RecLine ws, r, "30 ScreenUpdating", Err.Number, Err.Description, "", okN, fails

    Err.Clear
    Application.DisplayAlerts = False
    Application.DisplayAlerts = True
    RecLine ws, r, "31 DisplayAlerts", Err.Number, Err.Description, "", okN, fails

    Err.Clear
    Set d = CreateObject("Scripting.Dictionary")
    d.Add "a", 1
    t = d.Count
    RecLine ws, r, "32 Scripting.Dictionary", Err.Number, Err.Description, t, okN, fails

    Err.Clear
    v = Split("가|나|다", "|")
    t = Join(v, ",") & " " & InStrRev("a@b@c", "@") & " " & ChrW(44032) & AscW("가") & " " & Round(2.567, 1) & " " & Weekday(DateSerial(2026, 11, 11))
    RecLine ws, r, "33 문자열·날짜 함수", Err.Number, Err.Description, t, okN, fails

    Err.Clear
    t = FormatNumber(1234.5, 1)
    RecLine ws, r, "34 FormatNumber", Err.Number, Err.Description, t, okN, fails

    Err.Clear
    ws.Range("D40:F40").Value = Array("가", "나", "다")
    v = ws.Range("D40:F40").Value
    t = IsArray(v) & " " & UBound(v, 2)
    RecLine ws, r, "35 범위에 배열로 쓰기·읽기", Err.Number, Err.Description, t, okN, fails

    Err.Clear
    t = ws.UsedRange.Row & "," & ws.UsedRange.Rows.Count & "," & ws.UsedRange.Columns.Count
    RecLine ws, r, "36 UsedRange", Err.Number, Err.Description, t, okN, fails

    Err.Clear
    Set tmp = Nothing
    Set tmp = wb.Worksheets.Add()
    tmp.Visible = 0
    tmp.Visible = -1
    Application.DisplayAlerts = False
    tmp.Delete
    Application.DisplayAlerts = True
    RecLine ws, r, "37 시트 숨기기·지우기", Err.Number, Err.Description, "", okN, fails

    Err.Clear
    ws.Activate
    ActiveWindow.DisplayGridlines = False
    RecLine ws, r, "38 ActiveWindow 눈금선 끄기", Err.Number, Err.Description, "", okN, fails

    Err.Clear
    ws.Range("D41").Select
    t = TypeName(Selection) & " " & Selection.Address
    RecLine ws, r, "39 Selection", Err.Number, Err.Description, t, okN, fails

    Err.Clear
    t = Application.Workbooks.Count
    RecLine ws, r, "40 Workbooks.Count", Err.Number, Err.Description, t, okN, fails

    Err.Clear
    t = InputBox("InputBox 점검: 아무 글자나 쓰고 확인을 누르세요", "점검", "가나다")
    RecLine ws, r, "41 InputBox", Err.Number, Err.Description, t, okN, fails

    Err.Clear
    ws.Columns(1).AutoFit
    On Error GoTo 0
    MsgBox "점검 끝 — 성공 " & okN & "개" & vbLf & "실패: " & fails & vbLf & vbLf & "'" & ws.Name & "' 시트에 자세한 결과가 있습니다."
End Sub

Sub RecLine(ws, r, nm, errNo, errDesc, extra, okN, fails)
    Dim res
    On Error Resume Next
    If errNo = 0 Then
        res = "OK"
        okN = okN + 1
    Else
        res = "실패 " & errNo & ": " & errDesc
        fails = fails & Left(nm, 2) & " "
    End If
    If Len(extra & "") > 0 Then res = res & "  [" & extra & "]"
    r = r + 1
    If Not ws Is Nothing Then
        ws.Cells(r, 1).Value = nm
        ws.Cells(r, 2).Value = res
        If errNo <> 0 Then ws.Cells(r, 2).Font.Color = RGB(196, 0, 0)
    End If
End Sub

End Sub

' (선택) 파일 고르기와 다른 파일 열기 점검
Sub HancellFileTest()
    Dim f, wb2, t
    On Error Resume Next
    f = Application.GetOpenFilename("엑셀 파일 (*.xlsx),*.xlsx")
    If Err.Number <> 0 Then
        MsgBox "GetOpenFilename 실패: " & Err.Description
        Exit Sub
    End If
    If VarType(f) = 11 Then
        MsgBox "파일을 고르지 않았습니다"
        Exit Sub
    End If
    Set wb2 = Nothing
    Set wb2 = Application.Workbooks.Open(f, 0, True)
    If wb2 Is Nothing Then
        MsgBox "Workbooks.Open 실패: " & Err.Description
        Exit Sub
    End If
    t = wb2.Worksheets.Count & "개 시트, 첫 시트 " & wb2.Worksheets(1).Name & ", A1=" & wb2.Worksheets(1).Cells(1, 1).Value
    wb2.Close False
    MsgBox "다른 파일 읽기 성공: " & t & vbLf & "(오류: " & Err.Number & " " & Err.Description & ")"
End Sub
