' 한셀 매크로 점검 — 한셀 매크로 편집기에서 어떤 기능이 되는지 확인합니다.
'
' 쓰는 법
'   1. 한셀에서 새 문서를 엽니다.
'   2. 매크로 편집기(도구 → 매크로 → 스크립트/VBA 편집기)에서 모듈을 하나 만들고
'      이 파일 내용을 전부 붙여넣습니다.
'   3. HancellCheck 를 실행합니다. ("점검을 시작합니다" 창이 떠야 합니다)
'   4. 끝나면 결과 창이 몇 개 뜹니다(결과는 '점검' 시트에도 적힙니다). 창이나 시트를 캡처해 주세요.
'   5. (선택) 점검 시트에 생긴 [단추 점검] 단추를 눌러 보고, HancellFileTest 도 실행해 보세요.
'
' 일부러 한셀과 엑셀이 모두 읽는 문법만 썼습니다 (변수 형식·$ 함수·IIf 없음).
' ===========================================================================

Sub HancellCheck()

Sub ShowLog(logS, okN, fails)
    Dim parts, i, chunk, k, total
    On Error Resume Next
    parts = Split(logS, vbLf)
    total = 0
    chunk = ""
    k = 1
    For i = 0 To UBound(parts)
        If Len(chunk) + Len(parts(i)) > 850 Then
            MsgBox chunk, 0, "점검 결과 " & k
            chunk = ""
            k = k + 1
        End If
        If parts(i) <> "" Then chunk = chunk & parts(i) & vbLf
    Next
    MsgBox chunk & vbLf & "성공 " & okN & "개 / 실패: " & fails, 0, "점검 결과 " & k & " (끝)"
End Sub

Sub RecLine(ws, r, nm, errNo, errDesc, extra, okN, fails, logS)
    Dim res
    On Error Resume Next
    If errNo = 0 Then
        res = "OK"
        okN = okN + 1
    Else
        res = "실패 " & errNo & ": " & errDesc
        fails = fails & Left(nm, 3) & " "
    End If
    If Len(extra & "") > 0 Then res = res & "  [" & extra & "]"
    logS = logS & nm & " — " & res & vbLf
    r = r + 1
    If Not ws Is Nothing Then
        ws.Cells(r, 1).Value = nm
        ws.Cells(r, 2).Value = res
        If errNo <> 0 Then ws.Cells(r, 2).Font.Color = RGB(196, 0, 0)
    Else
        ' 결과 시트가 아직 없으면 지금 보이는 시트의 K·L열에 적음 (중간에 멈춰도 남도록)
        ActiveSheet.Cells(r, 11).Value = nm
        ActiveSheet.Cells(r, 12).Value = res
    End If
End Sub

' (선택) 파일 고르기와 다른 파일 열기 점검
Sub HancellFileTest()
    Dim f, wb2, t, fd, how
    On Error Resume Next
    how = ""
    Err.Clear
    f = Application.GetOpenFilename("엑셀 파일 (*.xlsx),*.xlsx")
    If Err.Number = 0 And VarType(f) = 8 Then
        how = "GetOpenFilename"
    Else
        t = "GetOpenFilename: " & Err.Number & " " & Err.Description
        Err.Clear
        Set fd = Nothing
        Set fd = Application.FileDialog(3)
        If Not fd Is Nothing Then
            fd.AllowMultiSelect = False
            If fd.Show = -1 Then
                f = fd.SelectedItems(1)
                how = "FileDialog"
            End If
        End If
        If how = "" Then
            t = t & vbLf & "FileDialog: " & Err.Number & " " & Err.Description
            Err.Clear
            f = InputBox("파일 고르기 창을 쓸 수 없습니다." & vbLf & "열어 볼 .xlsx 파일의 전체 경로를 쓰세요 (예: C:\Users\...\양식.xlsx)", "파일 경로")
            If f = "" Then
                MsgBox t & vbLf & "(경로를 쓰지 않아 끝냅니다)"
                Exit Sub
            End If
            how = "경로 입력"
        End If
    End If
    Err.Clear
    Set wb2 = Nothing
    Set wb2 = Application.Workbooks.Open(f, 0, True)
    If wb2 Is Nothing Then
        MsgBox "파일 고르기: " & how & vbLf & "Workbooks.Open 실패: " & Err.Number & " " & Err.Description & vbLf & t
        Exit Sub
    End If
    t = wb2.Worksheets.Count & "개 시트, 첫 시트 " & wb2.Worksheets(1).Name & ", A1=" & wb2.Worksheets(1).Cells(1, 1).Value
    wb2.Close False
    MsgBox "파일 고르기: " & how & vbLf & "다른 파일 읽기 성공: " & t & vbLf & "(오류: " & Err.Number & " " & Err.Description & ")"
End Sub
